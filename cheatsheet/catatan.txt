# chmod
chmod 755 file      -> rwxr-xr-x (owner full, lain baca+exec)
chmod 644 file      -> rw-r--r-- (file biasa)
chmod 600 file      -> rw------- (private, buat ssh key)
chmod 700 dir       -> rwx------ (folder private)

OCTAL:
4=r, 2=w, 1=x
7=4+2+1=rwx, 6=4+2=rw-, 5=4+1=r-x

SIMBOLIK:
chmod u+x file      -> tambah exec buat user
chmod g-w file      -> hapus write buat group
chmod o=r file      -> other cuma r
chmod a+r file      -> all tambah read
chmod u=rw,go=r file-> user rw, group+other r

RECURSIVE & SPECIAL BIT:
chmod -R 755 dir    -> recursive
chmod +s file       -> setuid (4000) -> jalan sebagai owner
chmod g+s dir       -> setgid (2000) -> file baru ikut group folder
chmod +t dir        -> sticky bit (1000) -> cuma owner bisa hapus, contoh /tmp

# glob
GLOB = pola file yang di-expand sama bash SEBELUM perintah jalan
Beda sama regex! glob buat file, regex buat isi file

BASIC:
* -> semua karakter, kecuali hidden. ex: *.txt
? -> satu karakter. ex: file?.txt = file1.txt fileA.txt
[abc] -> satu dari a,b,c. ex: file[12].txt
[a-z] -> range. ex: [a-z]*.log
[!abc] atau [^abc] -> bukan a,b,c. ex: [!0-9]* = gak diawali angka
*.{jpg,png} -> semua jpg dan png (pakai brace)

BRACE EXPANSION (bukan glob, tapi mirip, urutan eksekusi lebih dulu):
{a,b,c} -> a b c
file{1..3}.txt -> file1.txt file2.txt file3.txt
file{a..c}.txt -> filea.txt fileb.txt filec.txt
{1..10..2} -> 1 3 5 7 9
a{1,2}b{3,4} -> a1b3 a1b4 a2b3 a2b4
mkdir -p project/{src,dist,tests}
touch file{1..100}.log

DOTFILES (hidden):
* -> gak include file.hidden
.* -> include hidden tapi juga. dan.. (bahaya rm)
shopt -s dotglob-> baru * include hidden
ls -d.* -> list hidden aja, -d biar gak masuk isinya

RECURSIVE ** :
shopt -s globstar
** -> semua file & folder recursive
**/*.txt -> semua.txt di semua subfolder
**/ -> cuma folder recursive
ls **/*.log

EXTGLOB (powerful):
shopt -s extglob
?(pat) -> 0 atau 1 kali pat
*(pat) -> 0 atau lebih pat
+(pat) -> 1 atau lebih pat
@(pat1|pat2) -> salah satu dari pat
!(pat) -> bukan pat

Contoh extglob:
ls!(*.tmp) -> semua kecuali.tmp
ls @(*.jpg|*.png) -> jpg atau png
ls!(*.bak|*.log)
rm!(*.sh) -> hapus semua kecuali.sh (hati-hati)

SHOPT PENTING BUAT GLOB:
shopt -s nullglob -> kalau gak ada yang match, hasil kosong (bukan string "*.txt")
shopt -s failglob -> kalau gak match, error (aman buat script)
shopt -s dotglob -> * include hidden
shopt -s nocaseglob -> * gak peduli besar kecil
shopt -s globstar -> aktifin **
shopt -> lihat semua status
shopt -s extglob nullglob globstar -> biasanya diaktifkan bareng

rm./* -> lebih aman dari rm * kalau ada file namanya -rf

echo *.txt -> debug lihat expand jadi apa

'*.txt' -> gak di-expand
*.txt -> di-expand
"$var"/*.txt -> var gak di-glob, tapi *.txt di-glob

# symbol
STREAM DASAR (3 file descriptor):
0 = stdin  -> input, default keyboard
1 = stdout -> output normal, default layar
2 = stderr -> output error, default layar

REDIRECT (ngatur aliran ke file):
> file          -> stdout timpa file (1> file)
>> file         -> stdout append ke file
< file          -> stdin dari file
2> file         -> stderr timpa file
2>> file        -> stderr append
&> file         -> stdout+stderr timpa file (bash)
&>> file        -> stdout+stderr append
> /dev/null     -> buang stdout, layar bersih
2> /dev/null    -> buang error doang
&> /dev/null    -> buang semua
2>&1            -> arahkan stderr ke tempat stdout lagi pergi
1>&2            -> arahkan stdout ke tempat stderr
3> file         -> bikin fd 3 custom

Contoh redirect:
ls /ada /tidakada > out.txt 2> err.txt -> out ke out.txt, error ke err.txt
ls /ada /tidakada > out.txt 2>&1 -> semua gabung ke out.txt
ls /ada /tidakada &> semua.txt -> sama, versi pendek
ls /ada /tidakada > /dev/null -> sembunyiin output normal
ls /ada /tidakada 2> /dev/null -> sembunyiin error doang

PIPE | (sambungin stdout kiri ke stdin kanan):
ls | grep txt     -> output ls jadi input grep
cat file | sort | uniq -c | sort -nr
ls /ada /tidakada 2>&1 | grep tidakada -> gabung dulu baru pipe error nya
ps aux | grep nginx | grep -v grep
echo "a b c" | xargs -n1 -> pecah per kata

PIPE MODERN:
|& -> pipe stdout+stderr langsung (sama kayak 2>&1 |)
ls /ada /tidakada |& grep tidakada

PEMISAH PERINTAH:
;  -> jalan urut, gagal pun lanjut: cd /tmp; ls; cd -
&& -> AND, lanjut kalau sukses (exit 0): mkdir a && cd a && touch file
|| -> OR, lanjut kalau gagal: cd mydir || mkdir mydir && cd mydir
&  -> jalan di background: sleep 10 &
&& dan || bisa gabung: [ -f file ] && echo ada || echo gak ada

GROUPING:
(command) -> jalan di subshell, env gak bocor ke luar
{ command; } -> jalan di shell ini, harus ada spasi dan ; di akhir
contoh: (cd /tmp && touch a) -> setelah itu kamu masih di folder lama
        { cd /tmp && touch a; } -> kamu pindah ke /tmp beneran

SUBSTITUSI:
$(command) -> output command jadi argumen: echo "hari $(date +%A)"
`command` -> sama kayak $(), tapi jadul, susah nesting
${var} -> variabel: ${HOME}/file
$((1+2)) -> hitung aritmatika

DASAR HIDUP MATI (wajib hafal 2 ini):
Ctrl+O -> Save (Write Out), Enter buat konfirmasi
Ctrl+X -> Keluar, kalau belum save ditanya Y/N
Ctrl+G -> Help lengkap

NAVIGASI:
panah -> gerak
Ctrl+F / Ctrl+B -> maju mundur 1 char (F forward B back)
Ctrl+P / Ctrl+N -> atas bawah (Prev Next)
Ctrl+A / Ctrl+E -> awal / akhir baris (kayak bash)
Ctrl+Y / Ctrl+V -> Page Up / Down
Alt+\ / Alt+/ -> awal / akhir file
Ctrl+_ -> Go to line number (Ctrl+Shift+-)

EDIT:
Backspace / Delete -> hapus
Ctrl+K -> Cut 1 baris (Kepotong)
Alt+6 -> Copy 1 baris (gak hapus)
Ctrl+U -> Paste (Uncut)
Ctrl+K terus Ctrl+U -> duplicate baris
Alt+A -> Mark / seleksi, lalu panah buat seleksi, Ctrl+K buat cut seleksi
Ctrl+J -> Justify / rapihin paragraf
Tab -> indent (atur di nanorc)

CARI & GANTI:
Ctrl+W -> Where is / cari, Enter, Alt+W cari lagi selanjutnya
Ctrl+\ -> Replace, isi yang dicari, Enter, isi ganti
Alt+R -> Regex mode pas Replace
Alt+C -> Case sensitive pas cari

UNDO REDO & INFO:
Alt+U -> Undo
Alt+E -> Redo (E lagi)
Ctrl+C -> Info posisi baris, kolom, char
Alt+D -> hitung kata/baris/char

FILE LAIN & MULTI BUFFER:
Ctrl+R -> Read / insert file lain ke posisi kursor
Ctrl+O lalu Ctrl+T -> browser file buat pilih nama save
Alt+< / Alt+> -> pindah buffer file sebelumnya / selanjutnya (kalau buka banyak file: nano file1 file2)
Ctrl+O Alt+P -> backup file asli jadi file~

TAMPILAN & OPSI:
nano -m -> aktifin mouse bisa klik & scroll (enak di WSL)
nano -B -> auto backup jadi file~
nano -c -> selalu tampilkan posisi kursor
nano -i -> auto indent
nano -A / -S -> smooth scroll

# mariadb
INSTALL & SERVICE (Ubuntu/Debian):
sudo apt install mariadb-server mariadb-client
sudo systemctl status mariadb
sudo systemctl start/stop/restart mariadb
sudo systemctl enable mariadb
sudo mysql_secure_installation -> set root pass, hapus anonymous, disable remote root
config: /etc/mysql/mariadb.conf.d/50-server.cnf
log: /var/log/mysql/error.log
my.cnf global: /etc/mysql/my.cnf, user: ~/.my.cnf

CONNECT:
mariadb -u root -p -> login root (mysql masih bisa, symlink)
mariadb -u ershad -p -> login user
mariadb -u ershad -p -h 127.0.0.1 -P 3306 -> host port
mariadb -u ershad -p -e "SHOW DATABASES;" -> eksekusi 1 query tanpa masuk shell
mycli -u root -p -> client alternatif enak ada autocomplete (pip install mycli)

DALAM SHELL MARIADB (akhiran ; wajib):
SHOW DATABASES;
CREATE DATABASE toko CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
DROP DATABASE toko;
USE toko;
SHOW TABLES;
DESCRIBE users; atau SHOW CREATE TABLE users\G
SHOW TABLE STATUS\G

CRUD DASAR:
CREATE TABLE users (id INT AUTO_INCREMENT PRIMARY KEY, nama VARCHAR(100), email VARCHAR(100) UNIQUE, created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP);
INSERT INTO users (nama,email) VALUES ('ershad','a@b.com');
SELECT * FROM users;
SELECT * FROM users WHERE nama LIKE 'ers%';
SELECT COUNT(*) FROM users;
UPDATE users SET nama='Ershad M' WHERE id=1;
DELETE FROM users WHERE id=1;
TRUNCATE users; -> kosongin cepat

USER & PRIVILEGE:
CREATE USER 'app'@'localhost' IDENTIFIED BY 'passKuat123';
CREATE USER 'app'@'%' IDENTIFIED BY 'pass'; -> % = remote
SHOW GRANTS FOR 'app'@'localhost';
GRANT ALL PRIVILEGES ON toko.* TO 'app'@'localhost';
GRANT SELECT,INSERT,UPDATE ON toko.* TO 'app'@'localhost';
REVOKE DELETE ON toko.* FROM 'app'@'localhost';
FLUSH PRIVILEGES;
DROP USER 'app'@'localhost';
SELECT User,Host FROM mysql.user;
ALTER USER 'app'@'localhost' IDENTIFIED BY 'passBaru';

BACKUP & RESTORE (paling penting):
# backup 1 db
mariadb-dump -u root -p toko > toko.sql
mysqldump -u root -p toko > toko.sql # nama lama, masih jalan

# backup semua db
mariadb-dump -u root -p --all-databases > all.sql

# backup cuma struktur
mariadb-dump -u root -p --no-data toko > struktur.sql

# backup cuma data
mariadb-dump -u root -p --no-create-info toko > data.sql

# restore
mariadb -u root -p toko < toko.sql
mariadb -u root -p < all.sql

# backup otomatis + gzip + tanggal
mariadb-dump -u root -p toko | gzip > toko-$(date +%F).sql.gz

TOOLS RESMI MARIADB (semua ada --help):
mariadb -> client shell (dulu mysql)
mariadb-admin / mysqladmin -> admin server
  mysqladmin -u root -p ping
  mysqladmin -u root -p status
  mysqladmin -u root -p processlist
  mysqladmin -u root -p variables
  mysqladmin -u root -p create toko
  mysqladmin -u root -p drop toko

mariadb-check / mysqlcheck -> cek & repair table
  mysqlcheck -u root -p --all-databases
  mysqlcheck -u root -p --auto-repair toko
  mysqlcheck -u root -p --optimize toko

mariadb-show / mysqlshow -> list db/table cepat tanpa masuk shell
  mysqlshow -u root -p
  mysqlshow -u root -p toko
  mysqlshow -u root -p toko users

mariadb-dump / mysqldump -> backup (di atas)
mariadb-import / mysqlimport -> import CSV cepat
  mysqlimport -u root -p --local toko users.csv

mariadb-binlog / mysqlbinlog -> baca binlog buat point-in-time recovery
  mysqlbinlog /var/log/mysql/mysql-bin.000001 > binlog.sql

mariadb-backup -> pengganti xtrabackup, buat hot backup innodb
  sudo mariabackup --backup --target-dir=/backup/ --user=root -p
